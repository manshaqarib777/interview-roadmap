// Project 5 (L385) — the enterprise assistant (L385).

async function authenticate(req) {
  // the SAML (L385) or the OIDC (L239) — the corporate ID (L385).
  const identity = await oidc.verify(req.token);     // L239, L385
  return { employeeId: identity.sub, groups: identity.groups };
}

// 2 · THE RBAC (L241) — the roles and the scopes (L262).
const roles = {
  employee: ['read:docs'],                          // L241
  manager:  ['read:docs', 'approve:refund'],        // L241, L324
  admin:    ['read:docs', 'write:index'],           // L241
};

// 3 · THE GROUNDING (L280) — the private RAG (L385).
async function answer(identity, query) {
  authorize(identity, 'read:docs');                 // L241
  const chunks = await internalRetrieve(query);     // the internal (L349, L385)

           citations: chunks.map((c) => c.source) };    // L192, L337
}

// 4 · THE AUDIT (L322) — the compliance record (L385).
